Singleton Creational
Only one instance — global access point
class DatabaseConnection {
    private static DatabaseConnection instance;
    private String url;

    private DatabaseConnection() {
        this.url = "jdbc:mysql://localhost/mydb";
        System.out.println("DB connected to: " + url);
    }

    public static DatabaseConnection getInstance() {
        if (instance == null) {
            instance = new DatabaseConnection();
        }
        return instance;
    }

    public void query(String sql) {
        System.out.println("Running: " + sql);
    }
}

// Usage
DatabaseConnection db1 = DatabaseConnection.getInstance();
DatabaseConnection db2 = DatabaseConnection.getInstance();
System.out.println(db1 == db2); // true — same instance
Prototype Creational
Clone existing objects instead of creating from scratch
class GameCharacter implements Cloneable {
    String name;
    int health;
    String weapon;

    public GameCharacter(String name, int health, String weapon) {
        this.name = name;
        this.health = health;
        this.weapon = weapon;
    }

    @Override
    public GameCharacter clone() {
        try {
            return (GameCharacter) super.clone();
        } catch (CloneNotSupportedException e) {
            return null;
        }
    }
}

// Usage

GameCharacter copy = original.clone();
copy.name = "Warrior2";
// original.name is still "Warrior"

Adapter Structural
Make incompatible interfaces work together via a wrapper
// Target interface (what client expects)
interface FajrTimeCity {
    String getFajrTime(String cityName);
}

// Adaptee (incompatible existing class)
class FajrTimeGeo {
    public String getFajrTime(double lat, double lon) {
        return "Fajr time is 5:39 AM";
    }
}

// Adapter — bridges the gap
class FajrTimeAdapter implements FajrTimeCity {
    private FajrTimeGeo geo;

    public FajrTimeAdapter(FajrTimeGeo geo) {
        this.geo = geo;
    }

    @Override
    public String getFajrTime(String cityName) {
        double lat = getLatitude(cityName);
        double lon = getLongitude(cityName);
        return geo.getFajrTime(lat, lon);
    }

    private double getLatitude(String city) { return 21.5433; }
    private double getLongitude(String city) { return 39.1728; }
}

// Usage — client only knows FajrTimeCity
FajrTimeCity service = new FajrTimeAdapter(new FajrTimeGeo());
System.out.println(service.getFajrTime("Jeddah"));

Composite Structural
Treat individual objects and groups uniformly via tree structure
interface FileSystemItem {
    int getSize();
    void print(String indent);
}

class File implements FileSystemItem {
    private String name;
    private int size;
    public File(String name, int size) {
        this.name = name; this.size = size;
    }
    public int getSize() { return size; }
    public void print(String indent) {
        System.out.println(indent + name + " (" + size + "kb)");
    }
}

class Folder implements FileSystemItem {
    private String name;
    private List<FileSystemItem> children = new ArrayList<>();
    public Folder(String name) { this.name = name; }
    public void add(FileSystemItem item) { children.add(item); }
    public int getSize() {
        return children.stream().mapToInt(FileSystemItem::getSize).sum();
    }
    public void print(String indent) {
        System.out.println(indent + name + "/");
        for (FileSystemItem c : children) c.print(indent + "  ");
    }
}

// Usage
Folder root = new Folder("root");
root.add(new File("readme.txt", 5));
Folder src = new Folder("src");
src.add(new File("Main.java", 20));
root.add(src);
root.print("");   // root/ > readme.txt > src/ > Main.java
System.out.println(root.getSize()); // 25

Flyweight Structural
Share intrinsic state across thousands of objects to save memory
// Flyweight — holds shared (intrinsic) state
class TreeType {
    String name, texture;
    public TreeType(String name, String texture) {
        this.name = name; this.texture = texture;
    }
    public void draw(int x, int y) {
        System.out.println("Drawing " + name + " at (" + x + "," + y + ")");
    }
}

// Factory — returns shared flyweights
class TreeFactory {
    private static Map<String, TreeType> types = new HashMap<>();

    public static TreeType getTreeType(String name, String texture) {
        return types.computeIfAbsent(name, k -> new TreeType(name, texture));
    }
}

// Context — holds unique (extrinsic) state
class Tree {
    int x, y;
    TreeType type;
    public Tree(int x, int y, String name, String texture) {
        this.x = x; this.y = y;
        this.type = TreeFactory.getTreeType(name, texture);
    }
    public void draw() { type.draw(x, y); }
}

// Usage — 1 million trees, only 2 TreeType objects
Tree t1 = new Tree(10, 20, "Oak", "oak_texture");
Tree t2 = new Tree(50, 60, "Oak", "oak_texture"); // reuses same TreeType
Tree t3 = new Tree(15, 25, "Pine", "pine_texture");
Proxy Structural
Substitute that controls access — validates, lazy-loads, or caches
interface SpeechToTextService {
    String convertSpeechToText(String audioFile);
}

class RealSpeechToText implements SpeechToTextService {
    public RealSpeechToText() {
        System.out.println("Loading heavy speech model...");
    }
    public String convertSpeechToText(String audioFile) {
        return "Transcription of: " + audioFile;
    }
}

// Protection Proxy — validates API key before delegating
class SpeechToTextProxy implements SpeechToTextService {
    private RealSpeechToText real;
    private String apiKey;

    public SpeechToTextProxy(String apiKey) {
        this.apiKey = apiKey;
        // real service NOT created yet (lazy)
    }

    private boolean isValid() {
        return apiKey != null && !apiKey.isEmpty();
    }

    public String convertSpeechToText(String audioFile) {
        if (!isValid()) {
            return "Access denied: invalid API key";
        }
        if (real == null) real = new RealSpeechToText(); // lazy init
        return real.convertSpeechToText(audioFile);
    }
}

// Usage
SpeechToTextService s = new SpeechToTextProxy("my-secret-key");
System.out.println(s.convertSpeechToText("audio.mp3"));

abstract class SupportHandler {
    protected SupportHandler next;
    public void setNext(SupportHandler next) { this.next = next; }

    public void handle(int level) {
        if (level == 2) System.out.println("Level2 handled ticket");

    public void handle(int level) {
        System.out.println("Level3 (manager) handled ticket");
    }
}

// Usage — build chain

Level3 l3 = new Level3();
l1.setNext(l2); l2.setNext(l3);

Command Behavioral
Encapsulate request as object — enables undo, queue, log
interface Command {
    void execute();
    void undo();
}

class TextEditor {
    private StringBuilder text = new StringBuilder();
    public void write(String s) { text.append(s); }
    public void deleteLast(int n) {
        text.delete(text.length() - n, text.length());
    }
    public String getText() { return text.toString(); }
}

class WriteCommand implements Command {
    private TextEditor editor;
    private String text;
    public WriteCommand(TextEditor e, String t) { editor=e; text=t; }
    public void execute() { editor.write(text); }
    public void undo() { editor.deleteLast(text.length()); }
}

// Invoker — holds history for undo
class EditorInvoker {
    private Stack<Command> history = new Stack<>();
    public void run(Command cmd) { cmd.execute(); history.push(cmd); }
    public void undo() { if (!history.isEmpty()) history.pop().undo(); }
}

// Usage
TextEditor editor = new TextEditor();
EditorInvoker invoker = new EditorInvoker();
invoker.run(new WriteCommand(editor, "Hello "));
invoker.run(new WriteCommand(editor, "World"));
System.out.println(editor.getText()); // Hello World
invoker.undo();
System.out.println(editor.getText()); // Hello
Interpreter Behavioral
Define grammar and interpret sentences in a language
// Simple math expression interpreter: e.g. "3 + 5"
interface Expression {
    int interpret();
}

class NumberExpr implements Expression {
    private int number;
    public NumberExpr(int n) { this.number = n; }
    public int interpret() { return number; }
}

class AddExpr implements Expression {
    private Expression left, right;
    public AddExpr(Expression l, Expression r) { left=l; right=r; }
    public int interpret() { return left.interpret() + right.interpret(); }
}

class SubtractExpr implements Expression {
    private Expression left, right;
    public SubtractExpr(Expression l, Expression r) { left=l; right=r; }
    public int interpret() { return left.interpret() - right.interpret(); }
}

// Usage — interpret "10 + 3 - 2"
Expression expr = new SubtractExpr(
    new AddExpr(new NumberExpr(10), new NumberExpr(3)),
    new NumberExpr(2)
);
System.out.println(expr.interpret()); // 11
Iterator Behavioral
Traverse a collection without exposing its internal structure
interface Iterator<T> {
    boolean hasNext();
    T next();
}

class StudentArray {
    private String[] students;
    private int size = 0;

    public StudentArray(int capacity) {
        students = new String[capacity];
    }
    public void add(String s) { students[size++] = s; }

    public Iterator<String> iterator() {
        return new Iterator<String>() {
            int index = 0;
            public boolean hasNext() { return index < size; }
            public String next() { return students[index++]; }
        };
    }
}

// Usage — uniform traversal regardless of structure
StudentArray arr = new StudentArray(10);
arr.add("Ali"); arr.add("Sara"); arr.add("Omar");

Iterator<String> it = arr.iterator();
while (it.hasNext()) {
    System.out.println(it.next());
}
Mediator Behavioral
Central hub — objects communicate through mediator, not directly
interface ChatMediator {
    void sendMessage(String msg, User sender);
}

class ChatRoom implements ChatMediator {
    private List<User> users = new ArrayList<>();
    public void addUser(User u) { users.add(u); }
    public void sendMessage(String msg, User sender) {
        for (User u : users) {
            if (u != sender) u.receive(sender.name + ": " + msg);
        }
    }
}

class User {
    String name;
    private ChatMediator mediator;
    public User(String name, ChatMediator m) {
        this.name = name; this.mediator = m;
    }
    public void send(String msg) { mediator.sendMessage(msg, this); }
    public void receive(String msg) { System.out.println(name + " received: " + msg); }
}

// Usage — no user knows about other users directly
ChatRoom room = new ChatRoom();
User ali = new User("Ali", room);
User sara = new User("Sara", room);
room.addUser(ali); room.addUser(sara);
ali.send("Hello!"); // Sara received: Ali: Hello!
Memento Behavioral
Save and restore object state without exposing internals
// Memento — stores snapshot
class EditorMemento {
    private final String text;
    public EditorMemento(String text) { this.text = text; }
    public String getText() { return text; }
}

// Originator — creates and restores from mementos
class Editor {
    private String text = "";
    public void type(String s) { text += s; }
    public EditorMemento save() { return new EditorMemento(text); }
    public void restore(EditorMemento m) { text = m.getText(); }
    public String getText() { return text; }
}

// Caretaker — manages history
class History {
    private Stack<EditorMemento> stack = new Stack<>();
    public void push(EditorMemento m) { stack.push(m); }
    public EditorMemento pop() { return stack.pop(); }
}

// Usage
Editor editor = new Editor();
History history = new History();

editor.type("Hello ");
history.push(editor.save()); // checkpoint
editor.type("World");
System.out.println(editor.getText()); // Hello World
editor.restore(history.pop());
System.out.println(editor.getText()); // Hello
Observer Behavioral
One-to-many: subject notifies all observers on state change
interface Observer { void update(Player player); }

interface Subject {
    void attach(Observer o);
    void detach(Observer o);
    void notifyObservers();
}

class Player implements Subject {
    private String name;
    private int currentPoints;
    private double sessionAvg;
    private List<Observer> observers = new ArrayList<>();

    public Player(String name) { this.name = name; }

    public void attach(Observer o) { observers.add(o); }
    public void detach(Observer o) { observers.remove(o); }
    public void notifyObservers() {
        for (Observer o : observers) o.update(this);
    }

    public void setCurrentPoints(int pts) {
        this.currentPoints = pts;
        if (currentPoints > sessionAvg) notifyObservers();
    }

    public String getName() { return name; }
    public int getCurrentPoints() { return currentPoints; }
}

class Fan implements Observer {
    private String fanName;
    public Fan(String fanName) { this.fanName = fanName; }
    public void update(Player p) {
        System.out.println("Player " + p.getName() +
            " exceeded session avg with " + p.getCurrentPoints() + " pts!");
    }
}

// Usage
Player player = new Player("Ronaldo");
player.attach(new Fan("Ahmed"));
player.attach(new Fan("Sara"));
player.setCurrentPoints(50); // both fans notified

Visitor Behavioral
Add new operations to objects without modifying their classes
interface DocumentElement {
    void accept(DocumentVisitor visitor);
}

class Paragraph implements DocumentElement {
    public String text;
    public Paragraph(String t) { text = t; }
    public void accept(DocumentVisitor v) { v.visit(this); }
}
class Image implements DocumentElement {
    public String src;
    public Image(String s) { src = s; }
    public void accept(DocumentVisitor v) { v.visit(this); }
}

interface DocumentVisitor {
    void visit(Paragraph p);
    void visit(Image img);
}

// New operation — HTML export — without changing Paragraph or Image
class HtmlExportVisitor implements DocumentVisitor {
    public void visit(Paragraph p) {
        System.out.println("<p>" + p.text + "</p>");
    }
    public void visit(Image img) {
        System.out.println("<img src='" + img.src + "'/>");
    }
}

// Usage
List<DocumentElement> doc = List.of(
    new Paragraph("Hello World"),
    new Image("photo.png")
);
DocumentVisitor exporter = new HtmlExportVisitor();
for (DocumentElement el : doc) el.accept(exporter);
